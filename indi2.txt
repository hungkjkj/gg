f_get_mode(arr) =>
    if array.size(arr) == 0
        [-1, 0]
    else
        int[] unique_vals = array.new_int()
        int[] counts = array.new_int()
        for i = 0 to array.size(arr) - 1
            val = array.get(arr, i)
            idx = array.indexof(unique_vals, val)
            if idx == -1
                array.push(unique_vals, val)
                array.push(counts, 1)
            else
                array.set(counts, idx, array.get(counts, idx) + 1)
        max_c = 0
        mode_v = -1
        for i = 0 to array.size(unique_vals) - 1
            c = array.get(counts, i)
            if c > max_c
                max_c := c
                mode_v := array.get(unique_vals, i)
        [mode_v, max_c]

// Hàm lấy ra UNIX Timestamp (ms) của giờ mục tiêu trong tương lai
f_get_next_timestamp(mode_val, tz) =>
    if mode_val < 0
        0
    else
        tgt_h = math.floor(mode_val / 100)
        tgt_m = mode_val % 100
        cur_y = year(time, tz)
        cur_m = month(time, tz)
        cur_d = dayofmonth(time, tz)
        
        tgt_time = timestamp(tz, cur_y, cur_m, cur_d, tgt_h, tgt_m, 0)
        
        // Nếu giờ mục tiêu của ngày hôm nay đã đi qua, thì cộng thêm 1 ngày để đẩy nó sang ngày mai
        if tgt_time <= time
            tgt_time := timestamp(tz, cur_y, cur_m, cur_d + 1, tgt_h, tgt_m, 0)
        tgt_time
