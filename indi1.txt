indicator("D-VP - Top 85% & 95% Only", overlay=true, max_boxes_count=500, max_labels_count=500)

// --- Cài đặt đầu vào ---
grpGrid = "Cài đặt Lưới Volume & Chu kỳ"
anchorInput = input.string("Tùy chỉnh (Reset mỗi N ngày)", title="Chu kỳ Reset Profile", options=["1 Tuần", "1 Tháng", "1 Năm", "2 Năm", "3 Năm", "5 Năm", "Tùy chỉnh (Reset mỗi N ngày)"], group=grpGrid, tooltip="Thời điểm hệ thống xóa toàn bộ hộp cũ và bắt đầu đo lường lại từ đầu. Chọn 'Tùy chỉnh' để cài đặt số ngày thủ công.")
customDays = input.int(10, title="Số ngày Reset (Dành cho Tùy chỉnh)", minval=1, group=grpGrid, tooltip="Nhập số ngày giao dịch để hệ thống đo lường và reset lại. Ví dụ: 10 ngày.")
modeGrid = input.string("Cố định (Ticks/Pips)", title="Chế độ chia lưới", options=["Cố định (Ticks/Pips)", "Tự động (Dựa trên lịch sử)"], group=grpGrid)
fixedTicks = input.int(20, title="Số Ticks/Points mỗi ô (Dành cho Cố định)", minval=1, group=grpGrid, tooltip="Xác định độ dày cứng của 1 ô. Ví dụ: 20 ticks.")
rowCount = input.int(50, title="Số lượng ô (Dành cho Tự động)", minval=5, group=grpGrid)
lookbackDays = input.int(5, title="Số ngày đo biên độ (Dành cho Tự động)", minval=1, group=grpGrid)
showText = input.bool(true, title="Hiển thị con số Volume")

// Cài đặt Reset Thời Gian (Time Decay)
grpTime = "Cài đặt Thời gian (Time Decay)"
timeoutHours = input.float(1, title="Số giờ không chạm để Reset vùng (Hours)", minval=0.01, step=0.5, group=grpTime, tooltip="Nếu xem chu kỳ Năm (dùng nến 1H, 4H), hãy tăng số giờ này lên tương ứng với khung giờ của nến.")

// Cài đặt Ngưỡng Thống Kê (Percentiles)
grpStats = "Cài đặt Ngưỡng Thống Kê (%)"
pct1 = input.float(15.0, title="Ngưỡng % Yếu (Xanh -> Vàng)", minval=1.0, maxval=99.0, group=grpStats)
pct2 = input.float(50.0, title="Ngưỡng % Trung bình (Vàng -> Cam)", minval=1.0, maxval=99.0, group=grpStats)
pct3 = input.float(75.0, title="Ngưỡng % Cao (Cam -> Đỏ)", minval=1.0, maxval=99.0, group=grpStats)
pct4 = input.float(85.0, title="Ngưỡng % Mạnh (Đỏ -> Tím)", minval=1.0, maxval=99.0, group=grpStats)

// Cài đặt Bậc Màu (Color Bands)
grpColors = "Cài đặt Màu Sắc"
colBlue   = input.color(color.new(color.blue, 85), title="Màu nhóm Yếu", group=grpColors)
colYellow = input.color(color.new(#17e908, 85), title="Màu nhóm Trung bình", group=grpColors)
colOrange = input.color(color.new(#f6ff00, 85), title="Màu nhóm Cao", group=grpColors)
colRed    = input.color(color.new(color.red, 50), title="Màu nhóm Mạnh", group=grpColors)
colPurple = input.color(color.new(color.purple, 50), title="Màu nhóm Đột biến (Top)", group=grpColors)

// Cài đặt Đường Trung Bình Xu Hướng (VWAP Nội bộ)
grpMA = "Cài đặt Đường Xu Hướng (VWAP Nội bộ)"
showMA = input.bool(true, title="Hiển thị đường VWAP hiện tại", group=grpMA)
maLength = input.int(1, title="Số nến VWAP", minval=1, group=grpMA)
colMAUp = input.color(color.new(color.green, 0), title="Màu VWAP Tăng", group=grpMA)
colMADown = input.color(color.new(color.red, 0), title="Màu VWAP Giảm", group=grpMA)

// Cài đặt VWAP HTF & Bands (Đã thêm dải SD)
grpVWAP_HTF = "Cài đặt Đường VWAP & Dải SD"
showVWAP_HTF = input.bool(true, title="Hiển thị đường VWAP HTF", group=grpVWAP_HTF)
vwapHTFLength = input.int(89, title="Chu kỳ VWAP (N)", minval=1, group=grpVWAP_HTF, tooltip="Nhập N. Ví dụ N=89 tức là lấy trung bình khối lượng 89 nến.")
vwapHTFTF = input.timeframe("60", title="Khung thời gian tính VWAP", group=grpVWAP_HTF, tooltip="Mặc định '60' là 1 Giờ.")
colVWAPUp = input.color(color.new(#00E5FF, 0), title="Màu VWAP Tăng (Neon Cyan)", group=grpVWAP_HTF) 
colVWAPDown = input.color(color.new(#FF007F, 0), title="Màu VWAP Giảm (Neon Pink)", group=grpVWAP_HTF) 

showBands = input.bool(true, title="Hiển thị Dải SD ±2", group=grpVWAP_HTF)
sdMultiplier = input.float(2.0, title="Hệ số SD (Mặc định: 2)", minval=0.1, step=0.5, group=grpVWAP_HTF)
colSDUp = input.color(color.new(#00E5FF, 0), title="Màu SD Trên (+2) Xanh Neon", group=grpVWAP_HTF)
colSDDown = input.color(color.new(#FF007F, 0), title="Màu SD Dưới (-2) Hồng Neon", group=grpVWAP_HTF)

// --- Khởi tạo Biến và Mảng ---
var int periodStartTime = na
var float currentBinSize = na 
var array<float> binPrices = array.new_float(0)
var array<float> binMaxVol = array.new_float(0)      
var array<int>   binLastTouch = array.new_int(0)

var array<float> recPrice = array.new_float(0)
var array<float> recVol = array.new_float(0)
var array<int>   recStart = array.new_int(0)
var array<int>   recEnd = array.new_int(0) 

var array<box> drawBoxes = array.new_box(0)
var array<label> drawLabels = array.new_label(0)

// --- Tính toán Biên độ ---
highestHigh = request.security(syminfo.tickerid, "D", ta.highest(high, lookbackDays))
lowestLow = request.security(syminfo.tickerid, "D", ta.lowest(low, lookbackDays))

// --- Xác định Chu Kỳ Reset ---
bool isNewPeriod = false
var int dayCount = 0

if anchorInput == "Tùy chỉnh (Reset mỗi N ngày)"
    if ta.change(time("D")) != 0
        dayCount += 1
    if dayCount >= customDays
        isNewPeriod := true
        dayCount := 0
else if anchorInput == "1 Tuần"
    isNewPeriod := ta.change(time("W")) != 0
else if anchorInput == "1 Tháng"
    isNewPeriod := ta.change(time("M")) != 0
else if anchorInput == "1 Năm"
    isNewPeriod := ta.change(year) != 0
else if anchorInput == "2 Năm"
    isNewPeriod := ta.change(year) != 0 and (year % 2 == 0)
else if anchorInput == "3 Năm"
    isNewPeriod := ta.change(year) != 0 and (year % 3 == 0)
else if anchorInput == "5 Năm"
    isNewPeriod := ta.change(year) != 0 and (year % 5 == 0)

if isNewPeriod
    periodStartTime := time
    array.clear(binPrices)
    array.clear(binMaxVol)
    array.clear(binLastTouch) 
    array.clear(recPrice)
    array.clear(recVol)
    array.clear(recStart)
    array.clear(recEnd)
    
    if modeGrid == "Cố định (Ticks/Pips)"
        currentBinSize := fixedTicks * syminfo.mintick
    else
        float rangeSize = highestHigh[1] - lowestLow[1]
        if rangeSize <= 0 or na(rangeSize)
            rangeSize := syminfo.mintick * 100 
        float rawBinSize = rangeSize / rowCount
        currentBinSize := math.max(syminfo.mintick, math.round(rawBinSize / syminfo.mintick) * syminfo.mintick)

if na(periodStartTime)
    periodStartTime := time
    if modeGrid == "Cố định (Ticks/Pips)"
        currentBinSize := fixedTicks * syminfo.mintick
    else
        float rangeSize = highestHigh - lowestLow
        if rangeSize <= 0 or na(rangeSize)
            rangeSize := syminfo.mintick * 100 
        float rawBinSize = rangeSize / rowCount
        currentBinSize := math.max(syminfo.mintick, math.round(rawBinSize / syminfo.mintick) * syminfo.mintick)

// --- Xác định Khung thời gian ngầm (Dynamic TF) linh hoạt theo số ngày ---
string dynamicTF = "240"
if anchorInput == "1 Tuần"
    dynamicTF := "1"
else if anchorInput == "1 Tháng"
    dynamicTF := "5"
else if anchorInput == "1 Năm"
    dynamicTF := "60"
else if anchorInput == "2 Năm"
    dynamicTF := "120"
else if anchorInput == "3 Năm"
    dynamicTF := "240"
else if anchorInput == "5 Năm"
    dynamicTF := "240"
else if anchorInput == "Tùy chỉnh (Reset mỗi N ngày)"
    if customDays <= 7
        dynamicTF := "1"
    else if customDays <= 31
        dynamicTF := "5"
    else if customDays <= 180
        dynamicTF := "60"
    else
        dynamicTF := "240"

// --- Lấy dữ liệu Lower TF ngầm ---
arr_high_ltf  = request.security_lower_tf(syminfo.tickerid, dynamicTF, high)
arr_low_ltf   = request.security_lower_tf(syminfo.tickerid, dynamicTF, low)
arr_vol_ltf   = request.security_lower_tf(syminfo.tickerid, dynamicTF, volume)
arr_time_ltf  = request.security_lower_tf(syminfo.tickerid, dynamicTF, time)

// --- Logic Ghi Nhận Xung Lực Liên Tục ---
if array.size(arr_high_ltf) > 0 and not na(currentBinSize)
    float binSize = currentBinSize 
    
    float secPerBar = timeframe.in_seconds(timeframe.period)
    int timeoutBars = math.max(1, int(math.round((timeoutHours * 3600) / secPerBar)))
    
    for i = 0 to array.size(arr_high_ltf) - 1
        float h_ltf = array.get(arr_high_ltf, i)
        float l_ltf = array.get(arr_low_ltf, i)
        float v_ltf = array.get(arr_vol_ltf, i)
        int   t_ltf = array.get(arr_time_ltf, i) 
        
        float minPriceBin = math.round(l_ltf / binSize) * binSize
        float maxPriceBin = math.round(h_ltf / binSize) * binSize
        
        float startBin = math.min(minPriceBin, maxPriceBin)
        float endBin   = math.max(minPriceBin, maxPriceBin)
        
        int steps = math.round((endBin - startBin) / binSize)
        int actualNumBins = steps + 1 
        float volPerBin = v_ltf / actualNumBins
        
        for step = 0 to steps
            float price = startBin + (step * binSize)
            int binIndex = array.indexof(binPrices, price)
            
            if binIndex == -1
                array.push(binPrices, price)
                array.push(binMaxVol, volPerBin)
                array.push(binLastTouch, bar_index) 
                
                array.push(recPrice, price)
                array.push(recVol, volPerBin)
                array.push(recStart, t_ltf)
                array.push(recEnd, 0)
            else
                int lastTouchBar = array.get(binLastTouch, binIndex) 
                
                if (bar_index - lastTouchBar) >= timeoutBars
                    for k = array.size(recPrice) - 1 to 0
                        if array.get(recPrice, k) == price and array.get(recEnd, k) == 0
                            array.set(recEnd, k, t_ltf)
                            break
                    
                    array.set(binMaxVol, binIndex, volPerBin)
                    array.set(binLastTouch, binIndex, bar_index)
                    
                    array.push(recPrice, price)
                    array.push(recVol, volPerBin)
                    array.push(recStart, t_ltf)
                    array.push(recEnd, 0)
                else
                    float currentVol = array.get(binMaxVol, binIndex)
                    float updatedVol = currentVol + volPerBin
                    
                    array.set(binMaxVol, binIndex, updatedVol)
                    array.set(binLastTouch, binIndex, bar_index)
                    
                    for k = array.size(recPrice) - 1 to 0
                        if array.get(recPrice, k) == price and array.get(recEnd, k) == 0
                            array.set(recVol, k, updatedVol)
                            break

// --- Tính Toán Phân Vị và Vẽ Biểu Đồ ---
if barstate.islast 
    if array.size(recVol) > 0
        float val1 = array.percentile_linear_interpolation(recVol, pct1)
        float val2 = array.percentile_linear_interpolation(recVol, pct2)
        float val3 = array.percentile_linear_interpolation(recVol, pct3) 
        float val4 = array.percentile_linear_interpolation(recVol, pct4) 
                
        for b in drawBoxes
            box.delete(b)
        array.clear(drawBoxes)
        
        for l in drawLabels
            label.delete(l)
        array.clear(drawLabels)
                
        float binHalfHeight = currentBinSize / 2
        int totalRecords = array.size(recPrice)
        
        if totalRecords > 0
            for i = 0 to totalRecords - 1
                float v = array.get(recVol, i)
                
                if v >= val3
                    float p = array.get(recPrice, i)
                    int startT = array.get(recStart, i)
                    int endT = array.get(recEnd, i)
                    
                    if endT == 0
                        endT := time 
                        
                    color finalColor = na
                    if v < val4
                        finalColor := colRed      
                    else
                        finalColor := colPurple   
                    
                    string volText = ""
                    if showText
                        volText := str.tostring(v, format.volume)
                        
                    box newBox = box.new(left=startT, top=p + binHalfHeight, right=endT, bottom=p - binHalfHeight, xloc=xloc.bar_time, border_color=color.new(finalColor, 85), bgcolor=finalColor, text=volText, text_color=color.white, text_size=size.tiny, text_halign=text.align_right) 
                    array.push(drawBoxes, newBox)
                    
                    int centerT = int((startT + endT) / 2)
                    label newLbl = label.new(x=centerT, y=p, xloc=xloc.bar_time, text="", style=label.style_none, tooltip="Volume: " + str.tostring(v, format.volume))
                    array.push(drawLabels, newLbl)

// --- Vẽ VWAP Hiện Tại (Nội bộ) ---
vwapValue = math.sum(hl2 * volume, maLength) / math.sum(volume, maLength)
maTrendColor = vwapValue >= vwapValue[1] ? colMAUp : colMADown
plot(showMA ? vwapValue : na, color=maTrendColor, title="VWAP Internal", linewidth=2, style=plot.style_line)

// --- Tính toán VWAP Cố Định Khung Thời Gian (HTF VWAP) & Standard Deviation ---
vwapHTFValue = ta.vwma(hlc3, vwapHTFLength)
// Tính phương sai theo khối lượng (Variance)
varianceHTFValue = ta.vwma(math.pow(hlc3 - vwapHTFValue, 2), vwapHTFLength)
// Suy ra độ lệch chuẩn (Standard Deviation)
stdevHTFValue = math.sqrt(varianceHTFValue)

// Rút dữ liệu từ khung lớn (HTF)
[htf_vwap, htf_stdev] = request.security(syminfo.tickerid, vwapHTFTF, [vwapHTFValue, stdevHTFValue])

// Đổi màu đường VWAP chính
vwapHTFTrendColor = htf_vwap >= htf_vwap[1] ? colVWAPUp : colVWAPDown

// Vẽ đường VWAP HTF
plot(showVWAP_HTF ? htf_vwap : na, color=vwapHTFTrendColor, title="VWAP HTF", linewidth=2, style=plot.style_line)

// --- Vẽ 2 Dải SD (Bands) ---
upperBand = htf_vwap + (htf_stdev * sdMultiplier)
lowerBand = htf_vwap - (htf_stdev * sdMultiplier)

// Vẽ dải SD (Dùng kiểu circles (chấm tròn) để tạo hiệu ứng nét đứt mô phỏng)
plot(showVWAP_HTF and showBands ? upperBand : na, color=colSDUp, title="VWAP Upper +2SD (Nét đứt)", linewidth=1, style=plot.style_circles)
plot(showVWAP_HTF and showBands ? lowerBand : na, color=colSDDown, title="VWAP Lower -2SD (Nét đứt)", linewidth=1, style=plot.style_circles)
